/// `by_default = true` → BY DEFAULT semantics (explicit values accepted).

### Parser change

In `parse_generated_column_constraint` (currently at `ddl.rs:750`):

```rust
fn parse_generated_column_constraint(p: &mut Parser) -> Result<ColumnConstraint, DbError> {
    p.advance(); // GENERATED

    // GENERATED BY DEFAULT AS IDENTITY
    if p.eat_ident_ci("by") {
        if !p.eat_ident_ci("default") {
            return Err(DbError::ParseError {
                message: "expected DEFAULT after BY in GENERATED BY DEFAULT".into(),
                position: Some(p.current_pos()),
            });
        }
        p.expect(&Token::As)?;
        if !p.eat_ident_ci("identity") {
            return Err(DbError::ParseError {
                message: "expected IDENTITY after GENERATED BY DEFAULT AS".into(),
                position: Some(p.current_pos()),
            });
        }
        parse_identity_options(p)?; // consumes optional (START WITH ... INCREMENT BY ...)
        return Ok(ColumnConstraint::GeneratedIdentity { by_default: true });
    }

    // GENERATED ALWAYS AS ...
    if !p.eat_ident_ci("always") {
        return Err(DbError::ParseError {
            message: "expected ALWAYS or BY DEFAULT after GENERATED".into(),
            position: Some(p.current_pos()),
        });
    }
    p.expect(&Token::As)?;

    // GENERATED ALWAYS AS IDENTITY  (not an expression — identity column)
    if p.eat_ident_ci("identity") {
        parse_identity_options(p)?;
        return Ok(ColumnConstraint::GeneratedIdentity { by_default: false });
    }

    // GENERATED ALWAYS AS (expr) STORED|VIRTUAL  (existing computed column path)
    p.expect(&Token::LParen)?;
    let expr = parse_expr(p)?;
    p.expect(&Token::RParen)?;
    let kind = if p.eat_ident_ci("stored") {
        GeneratedColumnKind::Stored
    } else {
        let _ = p.eat_ident_ci("virtual");
        GeneratedColumnKind::Virtual
    };
    Ok(ColumnConstraint::Generated { expr, kind })
}

/// Parses and discards the optional identity options clause.
/// `(START WITH n INCREMENT BY n MINVALUE n MAXVALUE n NO CYCLE CACHE n)`
fn parse_identity_options(p: &mut Parser) -> Result<(), DbError> {
    if p.eat(&Token::LParen) {
        // Consume tokens until the matching ')' — simple depth-0 scan
        let mut depth = 1usize;
        while depth > 0 {
            match p.peek() {
                Token::LParen => { depth += 1; p.advance(); }
                Token::RParen => { depth -= 1; p.advance(); }
                Token::Eof => return Err(DbError::ParseError {
                    message: "unexpected EOF in identity options".into(),
                    position: None,
                }),
                _ => { p.advance(); }
            }
        }
    }
    Ok(())
}
```

// crates/axiomdb-catalog/src/schema_table.rs

/// Identifies whether a column is a SQL:2003 identity column.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
pub enum IdentityKind {
    #[default]
    None = 0,
    Always = 1,
    ByDefault = 2,
}

impl From<IdentityKind> for u8 {
    fn from(k: IdentityKind) -> u8 { k as u8 }
}
impl TryFrom<u8> for IdentityKind {
    type Error = ();
    fn try_from(b: u8) -> Result<Self, ()> {
        match b { 1 => Ok(Self::Always), 2 => Ok(Self::ByDefault), _ => Ok(Self::None) }
    }
}

### Insert enforcement

In `assign_auto_increment` (insert_helpers.rs), add a guard for `IdentityKind::Always` BEFORE the existing `is_auto_trigger` check:

```rust
fn assign_auto_increment(/* ... */) -> Result<(), DbError> {
    let Some(ai_col) = schema_cols.iter().position(|c| c.auto_increment) else {
        return Ok(());
    };

    // GENERATED ALWAYS AS IDENTITY: reject any explicit non-NULL value
    if schema_cols[ai_col].identity_kind == IdentityKind::Always {
        match values.get(ai_col) {
            Some(Value::Null) | Some(Value::Int(0)) | Some(Value::BigInt(0)) | None => {}
            _ => {
                return Err(DbError::InvalidValue {
                    reason: format!(
                        "cannot insert explicit value into column '{}': \
                         column is GENERATED ALWAYS AS IDENTITY",
                        schema_cols[ai_col].name
                    ),
                });
            }
        }
    }

    // Existing logic: auto-generate when NULL or 0
    let is_auto_trigger = matches!(
        values.get(ai_col),
        Some(Value::Null) | Some(Value::Int(0)) | Some(Value::BigInt(0))
    );
    // ... rest unchanged
}
```
